CREATE OR REPLACE FUNCTION
public.get_public_booking_month_summary_internal(
  p_service_id UUID,
  p_location_id UUID,
  p_from DATE,
  p_to DATE,
  p_provider_id UUID DEFAULT NULL
)
RETURNS TABLE (
  calendar_date DATE,
  configured_count BIGINT,
  available_count BIGINT,
  booked_count BIGINT,
  status TEXT
)
LANGUAGE SQL
STABLE
SECURITY INVOKER
SET search_path = ''
AS $$
  WITH rule_base AS (
    SELECT
      ar.provider_id,
      ar.location_id,
      ar.weekday,
      ar.start_time,
      ar.end_time,
      ar.slot_duration_minutes,
      ar.timezone,
      ar.valid_from,

        ar.location_id,

      ) AS duration_minutes
    FROM public.public_booking_availability_rules ar
    JOIN public.public_booking_providers pbp
      ON pbp.provider_id = ar.provider_id
     AND pbp.active = TRUE
    JOIN public.profiles pr
      ON pr.id = ar.provider_id
     AND pr.active = TRUE
    JOIN public.clinic_locations l

    JOIN public.provider_locations pl

    JOIN public.provider_services ps

      )

  ),
  calendar_days AS (
    SELECT gs::DATE AS calendar_date
    FROM GENERATE_SERIES(

      p_to::TIMESTAMP,

    ) gs

      AND p_to IS NOT NULL
      AND p_to >= p_from
      AND p_to <= p_from + 31
  ),
  raw_slots AS (
    SELECT DISTINCT
      d.calendar_date,
      rb.provider_id,
      rb.location_id,
      p_service_id AS service_id,
      slot_start,
      slot_start
        + MAKE_INTERVAL(mins => rb.duration_minutes) AS slot_end
    FROM rule_base rb
    JOIN calendar_days d
      ON EXTRACT(DOW FROM d.calendar_date)::SMALLINT = rb.weekday
     AND (
       rb.valid_from IS NULL
       OR d.calendar_date >= rb.valid_from
     )
     AND (
       rb.valid_until IS NULL
       OR d.calendar_date <= rb.valid_until
     )
    CROSS JOIN LATERAL GENERATE_SERIES(
      (
        d.calendar_date + rb.start_time
      ) AT TIME ZONE rb.timezone,
      (
        (
          d.calendar_date + rb.end_time
        ) AT TIME ZONE rb.timezone
      ) - MAKE_INTERVAL(mins => rb.duration_minutes),
      MAKE_INTERVAL(mins => rb.slot_duration_minutes)
    ) AS gs(slot_start)
    WHERE rb.duration_minutes IS NOT NULL
  ),
  classified AS (
    SELECT
      rs.*,
      public.public_booking_slot_is_available(
        rs.provider_id,
        rs.location_id,
        rs.service_id,
        rs.slot_start,
        NULL
      ) AS is_available,
      EXISTS (
        SELECT 1
        FROM public.appointments a
        WHERE a.provider_id = rs.provider_id
          AND a.status NOT IN (
            'cancelado',
            'remarcado',
            'faltou'
          )
          AND rs.slot_start <
            COALESCE(
              a.end_at,
              a.start_at + INTERVAL '30 minutes'
            )
          AND rs.slot_end > a.start_at
      ) AS is_booked
    FROM raw_slots rs
  ),
  stats AS (
    SELECT
      d.calendar_date,
      COUNT(c.slot_start)::BIGINT AS configured_count,
      COUNT(c.slot_start)
        FILTER (WHERE c.is_available = TRUE)::BIGINT
        AS available_count,
      COUNT(c.slot_start)
        FILTER (WHERE c.is_booked = TRUE)::BIGINT
        AS booked_count
    FROM calendar_days d
    LEFT JOIN classified c
      ON c.calendar_date = d.calendar_date
    GROUP BY d.calendar_date
  )
  SELECT
    s.calendar_date,
    s.configured_count,
    s.available_count,
    s.booked_count,
    CASE
      WHEN s.available_count > 0
        THEN 'available'
      WHEN s.configured_count > 0
       AND s.available_count = 0
       AND s.booked_count > 0
        THEN 'sold_out'
      ELSE 'unavailable'
    END::TEXT AS status
  FROM stats s
  ORDER BY s.calendar_date;
$$;

'Resumo mensal seguro: vagas configuradas, livres e esgotadas para o calendario publico.';

COMMIT;

function shiftBookingMonth(
  value: string,
  delta: number,
) {
  const [year, month] =
    value.split('-').map(Number)

  const date =
    new Date(year, month - 1 + delta, 1)

  return [
    date.getFullYear(),
    String(date.getMonth() + 1).padStart(2, '0'),
  ].join('-')
}

function bookingMonthBounds(value: string) {
  const [year, month] =
    value.split('-').map(Number)

  const lastDay =
    new Date(year, month, 0).getDate()

  return {
    from:
      value + '-01',
    to:
      value +
      '-' +
      String(lastDay).padStart(2, '0'),
  }
}

function buildBookingCalendarMonth(value: string) {
  const [year, month] =
    value.split('-').map(Number)

  const first =
    new Date(year, month - 1, 1)

  const count =
    new Date(year, month, 0).getDate()

  const cells: Array<string | null> =
    Array.from(
      { length: first.getDay() },
      () => null,
    )

  for (let day = 1; day <= count; day += 1) {
    cells.push(
      value +
        '-' +
        String(day).padStart(2, '0'),
    )
  }

  return cells
}

function bookingMonthLabel(value: string) {
  const [year, month] =
    value.split('-').map(Number)

  const label =
    new Intl.DateTimeFormat(
      'pt-BR',
      {
        month: 'long',
        year: 'numeric',
        timeZone: 'America/Sao_Paulo',
      },
    ).format(
      new Date(year, month - 1, 1),
    )

  return (
    label.charAt(0).toUpperCase() +
    label.slice(1)
  )
}

function addBookingDays(
  value: string,
  days: number,
) {
  const [year, month, day] =
    value.split('-').map(Number)

  const date =
    new Date(year, month - 1, day)

  date.setDate(
    date.getDate() + days,
  )

  return [
    date.getFullYear(),
    String(date.getMonth() + 1).padStart(2, '0'),
    String(date.getDate()).padStart(2, '0'),
  ].join('-')
}
`
