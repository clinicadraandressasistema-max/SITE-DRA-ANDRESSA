   36: function asRecord(value: unknown): AnyRecord | null {
   37:   return typeof value === 'object' && value !== null
   38:     ? (value as AnyRecord)
   39:     : null
   40: }

   42: function records(value: unknown): AnyRecord[] {
   43:   if (!Array.isArray(value)) return []
   44: 
   45:   return value
   46:     .map(asRecord)
   47:     .filter(
   48:       (item): item is AnyRecord =>
   49:         item !== null,
   50:     )
   51: }

   53: function text(
   54:   item: AnyRecord | null | undefined,
   55:   keys: string[],
   56:   fallback = '',
   57: ) {
   58:   if (!item) return fallback
   59: 
   60:   for (const key of keys) {
   61:     const value = item[key]
   62: 
   63:     if (
   64:       value !== undefined &&
   65:       value !== null &&
   66:       String(value).trim()
   67:     ) {
   68:       return String(value)
   69:     }
   70:   }
   71: 
   72:   return fallback
   73: }
