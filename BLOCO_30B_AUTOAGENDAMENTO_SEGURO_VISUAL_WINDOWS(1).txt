function findBlockEnd(src, openIndex) {
  let depth = 0
  let quote = null
  let escaped = false
  let lineComment = false
  let blockComment = false

  for (let i = openIndex; i < src.length; i += 1) {
    const ch = src[i]
    const next = src[i + 1]

    if (lineComment) {
      if (ch === '\n') lineComment = false
      continue
    }

    if (blockComment) {
      if (ch === '*' && next === '/') {
        blockComment = false
        i += 1
      }
      continue
    }

    if (quote) {
      if (escaped) {
        escaped = false
        continue
      }
      if (ch === '\\') {
        escaped = true
        continue
      }
      if (ch === quote) quote = null
      continue
    }

    if (ch === '/' && next === '/') {
      lineComment = true
      i += 1
      continue
    }

    if (ch === '/' && next === '*') {
      blockComment = true
      i += 1
      continue
    }

    if (ch === "'" || ch === '"' || ch === '`') {
      quote = ch
      continue
    }

    if (ch === '{') depth += 1

    if (ch === '}') {
      depth -= 1
      if (depth === 0) return i + 1
    }
  }

  return -1
}

function replaceNamedFunction(src, name, replacement) {
  const re = new RegExp(`(?:export\\s+)?(?:async\\s+)?function\\s+${name}\\s*\\(`)
  const match = re.exec(src)
  if (!match) fail(`não encontrei a função ${name}`)

  let paren = 0
  let quote = null
  let escaped = false
  let closeParen = -1

  const firstParen = src.indexOf('(', match.index)

  for (let i = firstParen; i < src.length; i += 1) {
    const ch = src[i]

    if (quote) {
      if (escaped) escaped = false
      else if (ch === '\\') escaped = true
      else if (ch === quote) quote = null
      continue
    }

    if (ch === "'" || ch === '"' || ch === '`') {
      quote = ch
      continue
    }

    if (ch === '(') paren += 1
    if (ch === ')') {
      paren -= 1
      if (paren === 0) {
        closeParen = i
        break
      }
    }
  }

  if (closeParen < 0) fail(`assinatura inválida da função ${name}`)
  const open = src.indexOf('{', closeParen)
  if (open < 0) fail(`corpo não encontrado da função ${name}`)
  const end = findBlockEnd(src, open)
  if (end < 0) fail(`fim da função ${name} não encontrado`)

  return src.slice(0, match.index) + replacement.trimEnd() + src.slice(end)
}

function replaceIfAfterMarker(src, marker, replacement) {
  const markerIndex = src.indexOf(marker)
  if (markerIndex < 0) fail(`marcador não encontrado: ${marker}`)

  const ifIndex = src.indexOf('if (', markerIndex)
  if (ifIndex < 0) fail(`if não encontrado após ${marker}`)

  const open = src.indexOf('{', ifIndex)
  const end = findBlockEnd(src, open)
  if (end < 0) fail(`fim do bloco não encontrado após ${marker}`)

  return src.slice(0, markerIndex) + replacement.trimEnd() + src.slice(end)
}

function replaceActionBlock(src, action, replacement) {
  const needle = `if (action === '${action}')`
  const start = src.indexOf(needle)
  if (start < 0) fail(`ação ${action} não encontrada`)
  const open = src.indexOf('{', start)
  const end = findBlockEnd(src, open)
  if (end < 0) fail(`fim da ação ${action} não encontrado`)
  return src.slice(0, start) + replacement.trimEnd() + src.slice(end)
}

function replaceFormBeforeMarker(src, headingNeedle, marker, replacement) {
  const heading = src.indexOf(headingNeedle)
  if (heading < 0) fail(`título não encontrado: ${headingNeedle}`)
  const form = src.indexOf('<div className="form-grid two">', heading)
  const end = src.indexOf(marker, form)
  if (form < 0 || end < 0) fail(`formulário não encontrado em ${headingNeedle}`)
  return src.slice(0, form) + replacement.trimEnd() + '\n\n                      ' + src.slice(end)
}

  if (!edge.includes(actionLine)) fail('ponto para security gate não encontrado')

  const helper = `
    // BLOCO_30B_SECURITY_GATE
    const clientIp =
      req.headers.get('cf-connecting-ip') ||
      req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() ||
      'unknown'

    async function allowAttempt(
      kind: string,
      maxAttempts: number,
      windowMinutes: number,
    ) {
      const fingerprintHash = await sha256(\`\${kind}:\${clientIp}\`)
      const since = new Date(
        Date.now() - windowMinutes * 60 * 1000,
      ).toISOString()

      const { count, error: countError } = await admin
        .from('public_booking_security_attempts')
        .select('id', { count: 'exact', head: true })
        .eq('action', kind)
        .eq('fingerprint_hash', fingerprintHash)
        .gte('created_at', since)

      if (countError) throw countError
      if ((count || 0) >= maxAttempts) return false

      const { error: insertError } = await admin
        .from('public_booking_security_attempts')
        .insert({
          action: kind,
          fingerprint_hash: fingerprintHash,
        })

      if (insertError) throw insertError
      return true
    }

`

  edge = edge.replace(actionLine, actionLine + helper)
}
